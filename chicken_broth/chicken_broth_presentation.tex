\begin{frame}{Why Chicken Broth?}
  \begin{itemize}
    \item Rich in minerals and collagen
    \item Versatile base for soups, sauces, and grains
    \item Perfect for cold weather comfort
  \end{itemize}
\end{frame}

\begin{frame}{Ingredients}
  \begin{itemize}
    \item 1 whole chicken (or 2 lbs chicken parts)
    \item 3 carrots, chopped
    \item 3 celery stalks, chopped
    \item 1 large onion, quartered
    \item 4 garlic cloves, smashed
    \item 2 bay leaves
    \item 10 peppercorns
    \item Fresh parsley and thyme sprigs
    \item 12 cups cold water
    \item Salt to taste
  \end{itemize}
\end{frame}

\begin{frame}{Prep Tips}
  \begin{itemize}
    \item Rinse chicken and remove giblets
    \item Roughly chop vegetables---no need to peel
    \item Tie herbs with kitchen twine for easy removal
  \end{itemize}
\end{frame}

\begin{frame}{Cooking Steps}
  \begin{enumerate}
    \item Place chicken, vegetables, and aromatics in a large stockpot.
    \item Cover with cold water and bring to a gentle simmer.
    \item Skim foam during the first 20 minutes.
    \item Simmer on low for 2.5--3 hours, partially covered.
    \item Remove chicken and strain broth through a fine-mesh sieve.
  \end{enumerate}
\end{frame}

\begin{frame}{Finishing Touches}
  \begin{itemize}
    \item Cool quickly using an ice bath around the pot
    \item Refrigerate and remove solidified fat for a lighter broth
    \item Adjust seasoning with salt and fresh herbs before serving
  \end{itemize}
\end{frame}

\begin{frame}{Serving Ideas}
  \begin{itemize}
    \item Sip warm with a squeeze of lemon
    \item Use as the base for chicken noodle soup
    \item Cook grains like rice or quinoa for added flavor
    \item Freeze in portions for quick weeknight meals
  \end{itemize}
\end{frame}
