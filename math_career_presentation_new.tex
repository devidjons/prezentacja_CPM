\begin{frame}{Monoid and Mathematics and Programming}
  \textbf{Associative Property Definition:}
  \begin{itemize}
    \item An operation $\oplus$ is associative if: $(a \oplus b) \oplus c = a \oplus (b \oplus c)$
  \end{itemize}

  \vspace{0.5cm}
  \textbf{Maximum Operation Example:}
  \begin{itemize}
    \item $\max(\max(a, b), c) = \max(a, \max(b, c))$
    \item Example: $\max(\max(3, 7), 5) = \max(7, 5) = 7$
    \item Example: $\max(3, \max(7, 5)) = \max(3, 7) = 7$
  \end{itemize}

  \vspace{0.5cm}
  \textbf{Maximum of a List:}

  \texttt{maximum([3, 7, 5, 2, 9, 1]) = 9}
\end{frame}
