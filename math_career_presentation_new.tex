\begin{frame}{Monoidy: matematyka i~programowanie}
  \textbf{Definicja własności łączności działania:}
  \begin{itemize}
    \item Działanie $\oplus$ jest łączne, jeśli: $(a \oplus b) \oplus c = a \oplus (b \oplus c)$
  \end{itemize}

  \vspace{0.5cm}
  \textbf{Przykład z~funkcją maksimum:}
  \begin{itemize}
    \item $\max(\max(a, b), c) = \max(a, \max(b, c))$
    \item Przykład: $\max(\max(3, 7), 5) = \max(7, 5) = 7$
    \item Przykład: $\max(3, \max(7, 5)) = \max(3, 7) = 7$
  \end{itemize}

  \vspace{0.5cm}
  \textbf{Maksimum listy:}

  \texttt{maximum([3, 7, 5, 2, 9, 1]) = 9}
\end{frame}
